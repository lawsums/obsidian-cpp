# 队列板子

> 循环队列：定长数组实现（C 语言），`front/rear/size` 环形复用空间，判空判满靠 `size` 不用浪费一格。

```` tab
``` c
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

#define MAX_SIZE 100

// 循环队列结构体
typedef struct {
    int data[MAX_SIZE];
    int front;
    int rear;
    int size;
} Queue;

// 初始化队列
void QueueInit(Queue *q) {
    q->front = 0;
    q->rear = 0;
    q->size = 0;
}

// 判断队列是否为空
bool QueueIsEmpty(Queue *q) {
    return q->size == 0;
}

// 判断队列是否已满
bool QueueIsFull(Queue *q) {
    return q->size == MAX_SIZE;
}

// 入队操作
bool QueueEnqueue(Queue *q, int value) {
    if (QueueIsFull(q)) {
        printf("队列已满，无法入队！\n");
        return false;
    }
    
    q->data[q->rear] = value;
    q->rear = (q->rear + 1) % MAX_SIZE;
    q->size++;
    return true;
}

// 出队操作
bool QueueDequeue(Queue *q, int *value) {
    if (QueueIsEmpty(q)) {
        printf("队列为空，无法出队！\n");
        return false;
    }
    
    *value = q->data[q->front];
    q->front = (q->front + 1) % MAX_SIZE;
    q->size--;
    return true;
}

// 获取队头元素
bool QueueFront(Queue *q, int *value) {
    if (QueueIsEmpty(q)) {
        printf("队列为空！\n");
        return false;
    }
    
    *value = q->data[q->front];
    return true;
}

// 获取队列大小
int QueueSize(Queue *q) {
    return q->size;
}

// 清空队列
void QueueClear(Queue *q) {
    q->front = 0;
    q->rear = 0;
    q->size = 0;
}

// 打印队列
void QueuePrint(Queue *q) {
    if (QueueIsEmpty(q)) {
        printf("队列为空\n");
        return;
    }
    
    printf("队列元素: ");
    int index = q->front;
    for (int i = 0; i < q->size; i++) {
        printf("%d ", q->data[index]);
        index = (index + 1) % MAX_SIZE;
    }
    printf("\n");
}
````
