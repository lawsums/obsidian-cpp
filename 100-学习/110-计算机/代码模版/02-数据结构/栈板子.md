# 栈板子

> 定长数组栈（C 语言）：`StackCreate / StackPush / StackPop / StackPeek`。函数统一带 `Stack` 前缀，避免和库内其它板子的 `push / pop / peek` 同名冲突。

```` tab
``` c
#include <stdio.h>
#include <stdlib.h>

typedef struct stack {
    int top;
    unsigned capacity;
    int* array;
} Stack;

Stack* StackCreate(unsigned capacity) {
    Stack* stack = (Stack*)malloc(sizeof(Stack));
    stack->capacity = capacity;
    stack->top = -1;
    stack->array = (int*)malloc(stack->capacity * sizeof(int));
    return stack;
}

void StackPush(Stack* stack, int item) {
    if (stack->top == stack->capacity - 1) {
        printf("Stack overflow\n");
        return;
    }

    stack->array[++stack->top] = item;
}

void StackPop(Stack* stack) {
    if (stack->top == -1) {
        printf("Stack underflow\n");
        return;
    }

    stack->top--;
}

void StackPeek(Stack* stack) {
    if (stack->top == -1) {
        printf("Stack is empty\n");
        return;
    }

    printf("Top element is %d\n", stack->array[stack->top]);
}
````
