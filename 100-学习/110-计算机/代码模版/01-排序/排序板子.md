void Merge(int start, int mid, int end, vector<int>& nums) {
    if (start >= end) return;

    vector<int> tmp;
    int p1 = start, p2 = mid + 1;
    while (p1 <= mid && p2 <= end) {
        if (nums[p1] <= nums[p2]) {
            tmp.push_back(nums[p1++]);
        } else {
            tmp.push_back(nums[p2++]);
        }
    }
    while (p1 <= mid) tmp.push_back(nums[p1++]);
    while (p2 <= end) tmp.push_back(nums[p2++]);

    for (int i = 0; i < tmp.size(); i++) {
        nums[start + i] = tmp[i];
    }
}

void MergeSort(int start, int end, vector<int>& nums) {
    if (start >= end) return;
    int mid = (start + end) / 2;

    MergeSort(start, mid, nums);
    MergeSort(mid + 1, end, nums);
    Merge(start, mid, end, nums);
}

void MergeSort(vector<int>& nums) {
    MergeSort(0, nums.size() - 1, nums);
}

int Partition(int lo, int hi, vector<int>& nums) {

    int pivot = nums[lo];
    int i = lo + 1, j = hi; // 将这个位置暂时留给pivot
    while (i <= j) {
        while (i < hi && nums[i] <= pivot) i++;
        while (j > lo && nums[j] >= pivot) j--;
        if (i >= j) break;
        swap(nums[i], nums[j]);
    }

    swap(nums[lo], nums[j]); // 将pivot放到它应该待的位置
    return j;
}


void QuickSort(int lo, int hi, vector<int>& nums) {
    if (lo >= hi) return;

    int j = Partition(lo, hi, nums);
    QuickSort(lo, j - 1, nums);
    QuickSort(j + 1, hi, nums);
}

void QuickSort(vector<int>& nums) {
    QuickSort(0, nums.size() - 1, nums);
}

// 大顶堆下沉调整（保证 i 及子树满足大顶堆性质）
void HeapSink(vector<int>& nums, int i, int size) {
    while (2 * i + 1 < size) {
        int left = 2 * i + 1;
        int right = left + 1;
        int largest = left;
        if (right < size && nums[right] > nums[left]) largest = right;
        if (nums[i] >= nums[largest]) break;
        swap(nums[i], nums[largest]);
        i = largest;
    }
}

// 原地堆排序（升序）
void HeapSort(vector<int>& nums) {
    int n = (int)nums.size();
    if (n < 2) return;

    // 1. 建大顶堆：从最后一个非叶节点 (n/2 - 1) 开始下沉，O(n)
    for (int i = n / 2 - 1; i >= 0; --i) {
        HeapSink(nums, i, n);
    }
    // 2. 把堆顶（最大值）换到当前末尾，缩小堆范围后重新下沉
    for (int i = n - 1; i > 0; --i) {
        swap(nums[0], nums[i]);
        HeapSink(nums, 0, i);
    }
}

// 按 nums 的值升序排下标，不改变原数组 nums
int n = nums.size();
vector<int> indices(n);
std::iota(indices.begin(), indices.end(), 0);

sort(indices.begin(), indices.end(), [&](int a, int b) {
    return nums[a] < nums[b];
