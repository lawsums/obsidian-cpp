import inspect
import re

DEBUG = 1  # 提交时改成 0 即可关闭所有日志

def _get_var_names(stack_depth=2):
    """提取 LV(x, y, z) 括号中的原始变量名字符串"""
    try:
        stack = inspect.stack()
        frame = stack[stack_depth]
        s = frame.code_context[0].strip()
        # 匹配 LV(...) 内部的内容，支持内部带括号（如函数调用或元组）的简单匹配
        m = re.search(r"LV\s*\((.*)\)", s)
        if m:
            # 按照逗号分割变量名，但排除掉被嵌套括号包围的逗号
            content = m.group(1)
            # 简单的逗号分割（如果参数里有复杂的带逗号表达式，可用更高级的正则，通常够用）
            names = [name.strip() for name in content.split(",")]
            return names
        return []
    except:
        return []

def _fmt(obj):
    if isinstance(obj, list):
        if obj and isinstance(obj[0], list):
            return "\n[" + ", \n".join(str(x) for x in obj) + "]"
        else:
            return "[" + ", ".join(str(x) for x in obj) + "]"
    elif isinstance(obj, dict):
        return "{" + ", ".join(f"{k}: {v}" for k, v in obj.items()) + "}"
    elif isinstance(obj, tuple):
        return "(" + ", ".join(str(x) for x in obj) + ")"
    return str(obj)

if DEBUG:
    def LOG(x):
        print(f"[DEBUG] {x}")
    def LOG_V(name, val):
        print(f"[DEBUG] {name}: {_fmt(val)}")
    def LV(*args):
        # 获取调用者的变量名列表
        names = _get_var_names(stack_depth=2)
        # 构造打印字符串
        parts = []
        for i, val in enumerate(args):
            # 如果解析出的变量名数量不够，用 '?' 代替
            name = names[i] if i < len(names) else "?"
            parts.append(f"{name}={_fmt(val)}")
        # 用 C++ 类似的 " | " 拼接起来
        print(f"[DEBUG] {' | '.join(parts)}")
else:
    def LOG(x):
        pass
    def LOG_V(name, val):
        pass
    def LV(*args):
        pass

#include <iostream>
#include <vector>
#include <unordered_map>

#define TEST(func, ...) \
    func(__VA_ARGS__);  \
    std::cout << #func << " passed!\n"

// 1. 设置模式开关：提交时将下面的 1 改为 0 即可
#define DEBUG_MODE 1

#if DEBUG_MODE
    #define LOG(x) std::cout << "[DEBUG] " << x << std::endl
    
    // 条件打印：只有 cond 为真时才输出内容 x
    #define CLOG(cond, x) do { if(cond) std::cout << "[COND] " << x << std::endl; } while(0)

    // 多参数 LV 核心实现：支持 1-5 个参数
    #define GET_LV(_1, _2, _3, _4, _5, NAME, ...) NAME
    #define LV(...) std::cout << "[DEBUG] "; GET_LV(__VA_ARGS__, LV5, LV4, LV3, LV2, LV1)(__VA_ARGS__)

    #define LV1(x) std::cout << #x << "=" << (x) << std::endl
    #define LV2(x, ...) std::cout << #x << "=" << (x) << " | "; LV1(__VA_ARGS__)
    #define LV3(x, ...) std::cout << #x << "=" << (x) << " | "; LV2(__VA_ARGS__)
    #define LV4(x, ...) std::cout << #x << "=" << (x) << " | "; LV3(__VA_ARGS__)
    #define LV5(x, ...) std::cout << #x << "=" << (x) << " | "; LV4(__VA_ARGS__)
#else
    #define LOG(x)
    #define CLOG(cond, x)
    #define LV(...)
#endif

// 2. 提供对常用容器的打印支持 (保持原样)
template<typename T, std::size_t N>
std::ostream& operator<<(std::ostream& os, const std::array<T, N>& a) {
    os << "[";
    for (size_t i = 0; i < N; ++i) os << a[i] << (i == N - 1 ? "" : ", ");
    return os << "]";
}

template<typename T>
std::ostream& operator<<(std::ostream& os, const std::vector<T>& v) {
    os << "[";
    for (size_t i = 0; i < v.size(); ++i) os << v[i] << (i == v.size() - 1 ? "" : ", ");
    return os << "]";
}

template<typename K, typename V>
std::ostream& operator<<(std::ostream& os, const std::unordered_map<K, V>& m) {
    os << "{";
    bool first = true;
    for (const auto& [k, v] : m) {
        if (!first) os << ", ";
        os << k << ":" << v;
        first = false;
    }
    return os << "}";
}
