## 标准答案
```cpp
#include <iostream>
#include <sys/epoll.h>
#include <queue>
#include <chrono>
#include <functional>
#include <vector>
#include <unistd.h>
#include <cerrno>

using Timestamp = std::chrono::milliseconds;
using Clock = std::chrono::steady_clock;
using Callback = std::function<void()>;

struct TimerTask {
    // 记录时间和回调函数
    Timestamp startTime;
    Timestamp execTime;
    Callback cb_;

    bool operator>(const TimerTask &other) const {
        return execTime > other.execTime;
    }

    bool operator<(const TimerTask &other) const {
        return execTime < other.execTime;
    }
};

class Timer {
private:
    // 任务编号
    int task_id_ = 0;

    // 维护小根堆
    std::priority_queue<TimerTask, std::vector<TimerTask>, std::greater<>> pq;

    Timestamp get_current_ms() const {
        return std::chrono::duration_cast<Timestamp>(Clock::now().time_since_epoch());
    }

public:
    // 插入任务, 返回任务编号
    int addTask(int delay_ms, Callback cb) {
        // 获取过期时间
        Timestamp expire = get_current_ms() + Timestamp(delay_ms);

        // 加入最小堆中
        pq.push({get_current_ms(), expire, cb});

        return ++task_id_;
    }

    // 获取当前最近过期任务的过期时间(ms)
    int getNextExpireTime() {
        auto now = get_current_ms();
        // 判定是否有任务
        if (pq.empty()) return -1;

        TimerTask top = pq.top();
        // 如果当前任务已过期需要立刻执行
        if (top.execTime <= now) return 0;
        // 将gap转成int
        return (top.execTime - now).count();
    }

	// 处理过期任务
    void handleExpiredTasks() {
        auto now = get_current_ms();

        // 确认当前任务确实过期了
        while (!pq.empty() && pq.top().execTime <= now) {
            // 弹出任务, 执行任务
            auto task = pq.top(); pq.pop();
            task.cb_();
        }
    }
};

int main() {
    int epfd = epoll_create1(0);
    struct epoll_event events[10];
    Timer timer;

    // 添加任务
    int task1 = timer.addTask(4000, []() { std::cout << "Task1 (4s) 触发\n"; });
    timer.addTask(1000, []() { std::cout << "Task2 (1s) 触发！\n"; });

    while (true) {
        auto nextTime = timer.getNextExpireTime();
        if (nextTime == -1) {
            std::cout << "所有任务执行完毕" << '\n';
            break;
        }
        int nfds = epoll_wait(epfd, events, 10, nextTime);

        // Q: EINTR是什么？
        // A: epoll_wait 阻塞等待 IO 事件时，如果进程收到信号（比如按下 Ctrl+C、kill 发信号），内核会打断阻塞系统调用, EINTR就是打断信号
        if (nfds == -1 && errno != EINTR) break;

        // 到期自动执行
        timer.handleExpiredTasks();
    }

    close(epfd);
}
```


## TODO
1. 如何实现更新操作？
	 我们可以引入一个 `Map`（如 `std::unordered_map<int, TimerTask*>` 或数组）记录 `task_id` 到任务的映射，或者给任务加一个 `bool is_valid` 的标记：
	- **更新/删除时**：不真正去堆里删，而是通过 `task_id` 找到该任务，将其 `is_valid` 改为 `false`（如果是更新，则同时 `addTask` 一个新时间的新任务）。
	- **触发时（`handleExpiredTasks`）**：当遇到堆顶元素过期时，先检查它的 `is_valid`。如果已经是 `false`（失效了），直接 `pop` 掉不执行，继续看下一个。
	 **核心思想**：用 $O(1)$ 的内存空间标记换取 $O(\log N)$ 的延迟处理成本，避免破坏堆结构。

2. 如何用红黑树写一个定时器？红黑树的定时器性能好在哪里？
