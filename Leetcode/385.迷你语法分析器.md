## 0.1 385.迷你语法分析器

给定一个字符串 s 表示一个整数嵌套列表，实现一个解析它的语法分析器并返回解析的结果 `NestedInteger` 。

列表中的每个元素只可能是整数或整数嵌套列表

**示例 1：**

**输入：**s = "324",
**输出：**324
**解释：**你应该返回一个 NestedInteger 对象，其中只包含整数值 324。

**示例 2：**

**输入：**s = "[123,[456,[789]]]",
**输出：**[123,[456,[789]]]
**解释：**返回一个 NestedInteger 对象包含一个有两个元素的嵌套列表：
1. 一个 integer 包含值 123
2. 一个包含两个元素的嵌套列表：
    i.  一个 integer 包含值 456
    ii. 一个包含一个元素的嵌套列表
         a. 一个 integer 包含值 789

**提示：**

- `1 <= s.length <= 5 * 104`
- `s` 由数字、方括号 `"[]"`、负号 `'-'` 、逗号 `','`组成
- 用例保证 `s` 是可解析的 `NestedInteger`
- 输入中的所有值的范围是 `[-106, 106]`

## 0.3 Solution 
### 0.3.1 嵌套递归写法    
```cpp
#define UNDEFINE INT_MIN

class Solution {
private:
    NestedInteger ans;
    int where = 0;
    bool neg = false;
    string s;

public:
    NestedInteger deserialize(string s) {
        // 嵌套函数
        this->s = s;
        f(0, ans);
        return ans.getList()[0];
    }

    void f(int i, NestedInteger &nums) {
        int tmp = UNDEFINE;
        while (i < s.size() && s[i] != ']') {
            if (s[i] == '[') { 
                // 进入下一层嵌套
                NestedInteger new_num;
                f(i + 1, new_num);
                nums.add(new_num);
                // f执行之后全局索引变动了, 获取全局索引where
                i = where + 1;
            } 
            // 如果是数字加到tmp里面
            else if ('0' <= s[i] && s[i] <= '9') {
                if (UNDEFINE == tmp) {
                    tmp = s[i] - '0';
                } else {
                    tmp = tmp * 10 + (s[i] - '0');
                }
                i += 1;
            } 
            // 如果是逗号加入前面的数
            else if (s[i] == ',') {
                push(tmp, nums);
                i += 1;
            } else if (s[i] == '-') {
                neg = true;
                i += 1;
            }
        }

        // 遇到]或者结束了
        push(tmp, nums);
        // 修改全局索引
        where = i;
    }

    void push(int &num, NestedInteger &nums) {
        if (UNDEFINE != num) {
            nums.add(neg ? -num : num);
            num = UNDEFINE;
            neg = false;
        }
    }
};
```

### 0.3.2 使用stoi解析string
```csharp
class Solution {
private:
    NestedInteger ans;
    int where = 0;
    string s;

public:
    NestedInteger deserialize(string s) {
        this->s = s;
        f(0, ans);
        return ans.getList()[0];
    }

    void f(int i, NestedInteger &nums) {
        while (i < s.size() && s[i] != ']') {
            if (s[i] == '[') {
                NestedInteger new_num;
                f(i + 1, new_num);
                nums.add(new_num);
                i = where + 1;
            }
            // ✅ 数字或负号 → 直接读一整段字符串
            else if (isdigit(s[i]) || s[i] == '-') {
                int start = i;
                while (i < s.size() && (isdigit(s[i]) || s[i] == '-')) i++;
                string num_str = s.substr(start, i - start);
                nums.add(stoi(num_str)); // ✅ stoi 直接转
            }
            else {
                i++; // 逗号跳过
            }
        }
        where = i;
    }
};
```

### 0.3.3 栈写法
```csharp
class Solution {
public:
    NestedInteger deserialize(string s) {
        stack<NestedInteger> stk;
        for(int i=0,n=s.length();i<n;){
            NestedInteger cur;
            if(s[i]=='['){
                stk.push(cur);
                i++;
            }
            else if(s[i]!=']'){
                string num="";
                while(i<n&&s[i]!=','&&s[i]!=']'){
                    num.push_back(s[i++]);
                }
                cur.setInteger(stoi(num));
                if(i<n&&s[i]==',')i++;
                if(stk.size()>=1)stk.top().add(cur);
                else stk.push(cur);
            }
            else{
                if(stk.size()>1){
                    cur = stk.top();stk.pop();
                    stk.top().add(cur);
                }
                i++;
                if(i<n&&s[i]==',')i++;
            }
        }
        return stk.top();
    }
};
```
