## 0.1 297.二叉树的序列化与反序列化

序列化是将一个数据结构或者对象转换为连续的比特位的操作，进而可以将转换后的数据存储在一个文件或者内存中，同时也可以通过网络传输到另一个计算机环境，采取相反方式重构得到原数据。

请设计一个算法来实现二叉树的序列化与反序列化。这里不限定你的序列 / 反序列化算法执行逻辑，你只需要保证一个二叉树可以被序列化为一个字符串并且将这个字符串反序列化为原始的树结构。

**提示:** 输入输出格式与 LeetCode 目前使用的方式一致，详情请参阅 [LeetCode 序列化二叉树的格式](https://leetcode.cn/help-center/3812581/)。你并非必须采取这种方式，你也可以采用其他的方法解决这个问题。

### 0.3.1 字符流写法
```cpp

class Codec {
public:
    // 序列化：将二叉树转为字符串（前序遍历，# 表示空节点）
    string serialize(TreeNode* root) {
        // 使用 stringstream 避免频繁字符串拼接，提升性能
        stringstream ss;
        serializeHelper(root, ss);
        return ss.str();
    }

    // 反序列化：将字符串转回二叉树
    TreeNode* deserialize(string data) {
        if (data.empty()) return nullptr;
        // 分割字符串为节点值列表
        vector<string> nodes = split(data, ',');
        int idx = 0;
        return deserializeHelper(nodes, idx);
    }

private:
    // 序列化辅助函数：递归前序遍历，写入 stringstream
    void serializeHelper(TreeNode* root, stringstream& ss) {
        if (!root) {
            ss << "#,";  // 空节点用 # 表示，逗号分隔
            return;
        }
        // 根 -> 左 -> 右
        ss << root->val << ",";
        serializeHelper(root->left, ss);
        serializeHelper(root->right, ss);
    }

    // 反序列化辅助函数：递归构建树，idx 用引用保证全局递增
    TreeNode* deserializeHelper(vector<string>& nodes, int& idx) {
        // 边界保护：索引越界直接返回空
        if (idx >= nodes.size() || nodes[idx] == "#") {
            idx++;
            return nullptr;
        }
        // 构建当前节点
        TreeNode* node = new TreeNode(stoi(nodes[idx++]));
        node->left = deserializeHelper(nodes, idx);
        node->right = deserializeHelper(nodes, idx);
        return node;
    }

    // 简化的分割函数：按分隔符分割字符串
    vector<string> split(const string& s, char delim) {
        vector<string> res;
        stringstream ss(s);
        string item;
        // 直接用 getline 分割，避免手动遍历，代码更简洁
        while (getline(ss, item, delim)) {
            res.push_back(item);
        }
        return res;
    }
};

```

### 0.3.2 我的手写写法

```cpp
class Codec {
public:

    // Encodes a tree to a single string.
    string serialize(TreeNode* root) {
        // 转成vector<string>, join(vector<string>, ",")
        // 主要将nullptr也看作一个节点
        if (!root) return "NULL";
        string ans = to_string(root->val);
        ans += "," + serialize(root->left);
        ans += "," + serialize(root->right);
        return ans;
    }

    // Decodes your encoded data to tree.
    TreeNode* deserialize(string data) {
        TreeNode *root;
        vector<string> nodes = split(data, ',');
        // 从开头前序遍历构建树
        int idx = 0;
        return desHandler(nodes, idx);
    }

    TreeNode *desHandler(vector<string> &nodes, int &idx) {
        // 如果当前是NULL直接直接返回nullptr
        if (nodes[idx] == "NULL") {
            idx++;
            return nullptr;
        }
        
        // 否则就是一个新节点, 值为stoi(string)
        TreeNode *node = new TreeNode(stoi(nodes[idx++]));
        node->left = desHandler(nodes, idx);
        node->right = desHandler(nodes, idx);
        return node;
    }

    vector<string> split(string &s, char op) {
        vector<string> ans;
        string path;
        int n = s.size();
        for (int i = 0; i < n; ) {
            while (i < n && s[i] != op) {
                path.push_back(s[i++]);
            }
            // s[i] == op
            ans.push_back(path);
            i++;
            path = "";
        }
        if (!path.empty()) ans.push_back(path);
        return ans;
    }
};
```
