## 102.二叉树的层序遍历

给你二叉树的根节点 `root` ，返回其节点值的 **层序遍历** 。 （即逐层地，从左到右访问所有节点）。

**示例 1：**

![](images/102_tree1.jpg)

**输入：**root = [3,9,20,null,null,15,7]
**输出：**[[3],[9,20],[15,7]]

**示例 2：**

**输入：**root = [1]
**输出：**[[139.单词拆分]]

**示例 3：**

**输入：**root = []
**输出：**[]

**提示：**

- 树中节点数目在范围 `[0, 2000]` 内
- `-1000 <= Node.val <= 1000`

这题里面我只想让你对比一下 `list`, `vector`, `slice` 的不同操作效率上面的不同

## Solution 

### C++
```cpp
class Solution {
public:
    vector<vector<int>> levelOrder(TreeNode* root) {
        vector<vector<int>> ans;
        TreeNode* q[2001];
        int h = 0, t = 0; // 头指针h，尾指针t（初始都为0）
        
        if (root == nullptr) {
            return {};
        }
        q[t++] = root;

        // 队列为空时终止（h == t 表示空）
        while (h != t) {
            vector<int> tmp; // 存储当前层的节点值
            int l = t - h;   // 当前层的节点数量
            
            // 遍历当前层的所有节点
            for (int i = 0; i < l; ++i) {
                auto cur = q[h++]; // 出队：头指针后移
                tmp.push_back(cur->val);
                if (cur->left) q[t++] = cur->left;
                if (cur->right) q[t++] = cur->right;
            }
            ans.push_back(tmp); // 收集当前层结果
        }

        return ans;
    }
};
```

### Go
```go
func levelOrder(root *TreeNode) [][]int {
    // 处理边界：根节点为空时直接返回空结果
    if root == nil {
        return [][]int{}
    }

    var ans [][]int
    // 改用动态切片作为队列，无需固定长度，更符合Go习惯
    q := []*TreeNode{root}

    // 队列不为空时循环
    for len(q) > 0 {
        var tmp []int
        // 获取当前层的节点数量（关键：必须先记录，因为后续队列会变化）
        levelLen := len(q)
        
        // 遍历当前层的所有节点
        for i := 0; i < levelLen; i++ {
            // 取出队列头部节点（模拟出队）
            cur := q[0]
            // slice是以下结构, q = q[1:]的操作只是移动指针, len--, 复杂度为O(1)
            // type slice struct {
            //     ptr *[]byte // 指向底层数组的指针
            //     len int     // 切片长度
            //     cap int     // 切片容量
            // }
            q = q[1:]

            // 将当前节点值加入临时切片
            tmp = append(tmp, cur.Val)

            // 左子节点非空则入队
            if cur.Left != nil {
                q = append(q, cur.Left)
            }
            // 右子节点非空则入队
            if cur.Right != nil {
                q = append(q, cur.Right)
            }
        }
        // 将当前层的结果加入最终答案
        ans = append(ans, tmp)
    }

    return ans
}
```

### Python
```python
class Solution:
    def levelOrder(self, root: Optional[TreeNode]) -> List[List[int]]:
        if not root: return []
        dq = deque([root])
        ans = []
        while dq:
            tmp = []
            for i in range(len(dq)):
                cur = dq.popleft() # 提取出最老的元素
                tmp.append(cur.val)
                if cur.left: dq.append(cur.left)
                if cur.right: dq.append(cur.right)    
            ans.append(tmp)

        return ans
```
