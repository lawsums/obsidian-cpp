## 0.1 1000046.最大子矩阵

给定一个正整数、负整数和 0 组成的 N × M 矩阵，编写代码找出元素总和最大的子矩阵。

返回一个数组 `[r1, c1, r2, c2]`，其中 `r1`, `c1` 分别代表子矩阵左上角的行号和列号，`r2`, `c2` 分别代表右下角的行号和列号。若有多个满足条件的子矩阵，返回任意一个均可。

**注意：**本题相对书上原题稍作改动

**示例：**

**输入：**
`[    [-1,**0**],    [0,-1] ]`
**输出：**[0,1,0,1]
**解释：**输入中标粗的元素即为输出所表示的矩阵

**说明：**

- `1 <= matrix.length, matrix[0].length <= 200`

## 0.3 Solution 
```cpp
class Solution {
public:
    vector<int> getMaxMatrix(vector<vector<int>>& matrix) {
        int n = matrix.size();    // 行数
        int m = matrix[0].size();// 列数
        int maxSum = INT_MIN;
        vector<int> ans(4);      // 最终答案：[r1,c1,r2,c2]

        // 枚举 上边界
        for (int top = 0; top < n; ++top) {
            vector<int> colSum(m, 0); // 按列求和，长度 = 列数 m
            
            // 枚举 下边界
            for (int bottom = top; bottom < n; ++bottom) {
                // 把当前行 bottom 累加到列和数组
                for (int c = 0; c < m; ++c) {
                    colSum[c] += matrix[bottom][c];
                }

                // 关键：一维Kadane，返回 最大和、起始列、结束列
                auto [curSum, c1, c2] = kadane(colSum);

                // 更新全局最大
                if (curSum > maxSum) {
                    maxSum = curSum;
                    ans = {top, c1, bottom, c2};
                }
            }
        }
        return ans;
    }

    // 核心修改：返回 tuple(最大和, 起始下标, 结束下标)
    tuple<int, int, int> kadane(vector<int>& nums) {
        int maxCur = nums[0];
        int maxTotal = nums[0];
        int start = 0;
        int finalStart = 0;
        int finalEnd = 0;

        for (int i = 1; i < nums.size(); ++i) {
            if (maxCur + nums[i] < nums[i]) {
                maxCur = nums[i];
                start = i; // 新起点
            } else {
                maxCur += nums[i];
            }

            // 更新全局最大
            if (maxCur > maxTotal) {
                maxTotal = maxCur;
                finalStart = start;
                finalEnd = i;
            }
        }

        return {maxTotal, finalStart, finalEnd};
    }
};
```
