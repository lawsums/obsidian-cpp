## 3.2 第三天

### 3.2.1 实现万能 Makefile
```makefile
# 基础配置（保留原书的编译逻辑）
AS = ../z_tools/nask.exe
CC1 = ../z_tools/gcc.exe -m32 -fno-asynchronous-unwind-tables -s -O2 -c -Wall -Werror
GAS2NASK = ../z_tools/gas2nask.exe -a
OBJ2BIM = ../z_tools/obj2bim.exe
MAKEFONT = ../z_tools/makefont.exe
BIN2OBJ = ../z_tools/bin2obj.exe
BIM2HRB = ../z_tools/bim2hrb.exe
RULEFILE = ../z_tools/haribote/haribote.rul
EDIMG = ../z_tools/edimg.exe
IMGTOL = ../z_tools/imgtol.com
COPY = cmd /c copy
DEL = cmd /c del

# 自动检测当前目录下的.img文件（核心修复1：动态获取img文件名）
IMG_FILE := $(wildcard *.img)
ifeq ($(words $(IMG_FILE)),0)
$(error 错误：未找到任何.img文件！请先执行 make img 生成镜像文件)
endif
ifeq ($(words $(IMG_FILE)),1)
TARGET_IMG := $(IMG_FILE)
else
$(warning 警告：检测到多个.img文件，默认使用第一个：$(firstword $(IMG_FILE)))
TARGET_IMG := $(firstword $(IMG_FILE))
endif

# 原书的 img 目标（保留不动）
img:
	../z_tools/make.exe -r haribote.img

# 原书的 haribote.img 目标（保留不动，确保能编译出img）
haribote.img: ipl.bin
	$(EDIMG) imgin:../z_tools/fdimg0at.tek \
		wbinimg src:ipl.bin len:512 from:0 to:0 \
		imgout:haribote.img

# ipl.bin 编译目标（保留不动）
ipl.bin: ipl.nas
	$(AS) ipl.nas ipl.bin ipl.lst

# 修复后的 run 目标（核心修复2：加 cmd /c，用动态文件名）
run: img
	$(COPY) $(TARGET_IMG) ..\z_tools\qemu\fdimage0.bin
	../z_tools/make.exe -C ../z_tools/qemu

# 修复后的 clean 目标（之前的 del 问题也一并修复）
clean:
	$(DEL) ipl.bin
	$(DEL) ipl.lst

# 彻底清理目标（可选，清理img文件）
src_only: clean
	$(DEL) haribote.img

```
